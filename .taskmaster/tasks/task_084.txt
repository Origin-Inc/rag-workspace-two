class FormulaEngine {
  private columnFormulas: Map<string, ColumnFormula> = new Map();
  
  applyColumnFormula(column: string, formula: string) {

2. Add undo/redo hooks to formula evaluation:
```typescript
interface FormulaChange {
  type: 'cell' | 'column' | 'named_range';
  previous: string | null;
  current: string;
  affectedCells: CellReference[];
  timestamp: Date;
}

class UndoRedoStack {
  private undoStack: FormulaChange[] = [];
  private redoStack: FormulaChange[] = [];
  
  pushChange(change: FormulaChange) {
    this.undoStack.push(change);
    this.redoStack = []; // Clear redo on new change
  }
  
  undo(): FormulaChange | null {
    const change = this.undoStack.pop();
    if (change) {
      this.redoStack.push(change);
      return this.reverseChange(change);
    }
    return null;
  }
}
```

3. Implement named range registry:
```typescript
interface NamedRange {
  name: string;
  range: string; // e.g., 'A1:B10' or 'Sheet1!A:A'
  description?: string;
  createdBy: string;
  createdAt: Date;
  usedInFormulas: Set<string>;
}

class NamedRangeManager {
  private ranges: Map<string, NamedRange> = new Map();
  
  defineRange(name: string, range: string) {
    // Validate name (no spaces, valid identifier)
    if (!/^[a-zA-Z_][a-zA-Z0-9_]*$/.test(name)) {
      throw new Error('Invalid range name');
    }
    
    this.ranges.set(name, {
      name,
      range,
      createdAt: new Date(),
      usedInFormulas: new Set()
    });
    
    // Update HyperFormula instance
    this.hyperformula.addNamedExpression(name, range);
  }
}
```

4. Create formula versioning system:
```typescript
interface FormulaVersion {
  id: string;
  cellRef: string;
  formula: string;
  version: number;
  timestamp: Date;
  userId: string;
  changeReason?: string;
}

class FormulaHistory {
  private history: Map<string, FormulaVersion[]> = new Map();
  
  recordChange(cellRef: string, formula: string, userId: string) {
    const versions = this.history.get(cellRef) || [];
    const newVersion: FormulaVersion = {
      id: generateId(),
      cellRef,
      formula,
      version: versions.length + 1,
      timestamp: new Date(),
      userId
    };
    versions.push(newVersion);
    this.history.set(cellRef, versions);
  }
  
  getHistory(cellRef: string): FormulaVersion[] {
    return this.history.get(cellRef) || [];
  }
}
```
