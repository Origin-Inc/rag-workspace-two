3. **Implement Smart Formula Result Cache**
   ```typescript
   // app/services/spreadsheet/formula-cache.ts
   export class FormulaCache {
     private cache: LRUCache<string, CachedResult>;
     private dependencyGraph: Map<string, Set<string>>;
     
     constructor(maxSize = 10000) {
       this.cache = new LRUCache({ max: maxSize });
     }
     
     get(cellRef: string): CachedResult | null {
       const cached = this.cache.get(cellRef);
       if (cached && !this.isInvalidated(cellRef)) {
         return cached;
       }
       return null;
     }
     
     invalidateDependents(cellRef: string) {
       const dependents = this.dependencyGraph.get(cellRef) || new Set();
       for (const dependent of dependents) {
         this.cache.delete(dependent);
         this.invalidateDependents(dependent); // Recursive invalidation
       }
     }
   }
   ```

       setState(produce(draft => {

           draft.cells[row][col] = value;

6. **Implement Incremental Recalculation**
   ```typescript
   // app/services/spreadsheet/incremental-calculator.ts
   export class IncrementalCalculator {
     private dirtySet: Set<string> = new Set();
     private calculationOrder: string[] = [];
     
     markDirty(cellRef: string) {
       this.dirtySet.add(cellRef);
       // Mark all dependents as dirty
       const dependents = this.getDependents(cellRef);
       dependents.forEach(dep => this.dirtySet.add(dep));
     }
     
     async recalculate(): Promise<void> {
       // Sort dirty cells by dependency order
       const sortedCells = this.topologicalSort(this.dirtySet);
       
       // Calculate in batches for efficiency
       const batchSize = 100;
       for (let i = 0; i < sortedCells.length; i += batchSize) {
         const batch = sortedCells.slice(i, i + batchSize);
         await this.evaluateBatch(batch);
       }
       
       this.dirtySet.clear();
     }
   }
   ```

   // app/services/spreadsheet/memory-manager.ts
   export class MemoryManager {
     private cellStorage: Map<string, CompressedCell>;
     private compressionThreshold = 1000; // Compress cells with >1KB data
     
     storeCell(ref: string, value: any) {
       const size = this.calculateSize(value);
       if (size > this.compressionThreshold) {
         // Compress large values
         const compressed = this.compress(value);
         this.cellStorage.set(ref, { compressed: true, data: compressed });
       } else {
         this.cellStorage.set(ref, { compressed: false, data: value });
       }
     }
     
     // Implement virtual memory paging for cells not in viewport
     pageOutInactiveCells() {
       const inactiveCells = this.getInactiveCells();
       for (const cell of inactiveCells) {
         this.pageToIndexedDB(cell);
         this.cellStorage.delete(cell);
       }
     }
   }
